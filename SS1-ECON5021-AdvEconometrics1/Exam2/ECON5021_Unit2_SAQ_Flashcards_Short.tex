%===================== L8 =====================
\card{L8 Q1: $\mu$ minimizes $\E(y-a)^2$}
{add and subtract $\mu$; the cross term is 0.}
{$\E(y-a)^2=\E(y-\mu)^2+(\mu-a)^2$ (cross term $2(\mu-a)\E(y-\mu)=0$). Minimized at $a=\mu$. $\blacksquare$}

\card{L8 Q2: CEF is the best predictor}
{(a) definition, (b) LIE, (c) Card 1 trick with $h=m-g$.}
{(a) $\E(u|x)=m-m=0$.\quad (b) $\E(hu)=\E[h\,\E(u|x)]=0$.\\
(c) $\E(y-g)^2=\E u^2+2\E[u(m-g)]+\E(m-g)^2=\E u^2+\E(m-g)^2$, minimized at $g=m$. $\blacksquare$}

\card{L8 Q3: residual facts}
{FOC is $\sum x_{ik}e_i=0$; then $k=1$; average; Cov.}
{(a) $k=1$: $\sum e_i=0$.\quad (b) $\bar e=\bar y-\bar x'b=0$.\quad (c) $\widehat\Cov(x_k,e)=\tfrac1n\sum x_{ik}e_i-\bar x_k\bar e=0-0$. $\blacksquare$}

\card{L8 Q4: bivariate $b_2=\Cov/\Var$}
{factor out $n$; the $n$'s cancel; $2\times2$ inverse; row 2.}
{(a) $X'X=n\begin{bmatrix}1&\bar e\\ \bar e&\overline{ee}\end{bmatrix}$, $X'Y=n\begin{bmatrix}\bar w\\ \overline{ew}\end{bmatrix}$, so the $n$'s cancel in $\XXi X'Y$.\\
(b) Inverse $=\frac{1}{\overline{ee}-\bar e^2}\begin{bmatrix}\overline{ee}&-\bar e\\-\bar e&1\end{bmatrix}$; row 2: $b_2=\dfrac{\overline{ew}-\bar e\bar w}{\overline{ee}-\bar e^2}=\dfrac{\widehat\Cov(e,w)}{\widehat\Var(e)}$. $\blacksquare$}

%===================== L9 =====================
\card{L9 Q1: OLS is unbiased}
{substitute $Y$; A3; LIE.}
{$b=\beta+\XXi X'\veps\Rightarrow\E(b|X)=\beta+\XXi X'\underbrace{\E(\veps|X)}_0=\beta$;\; $\E b=\E[\E(b|X)]=\beta$. $\blacksquare$}

\card{L9 Q2: $\Var(b|X)=\sigma^2\XXi$}
{$b-\beta=A\veps$; $A\,\E(\veps\veps')A'$; $\sigma^2I$; cancel.}
{(a) $b=\beta+\XXi X'\veps$.\quad (b) $\Var(b|X)=\XXi X'\E(\veps\veps'|X)X\XXi$.\\
(c) $\Var(\veps|X)=\E(\veps\veps'|X)-0$ since $\E(\veps|X)=0$.\\
(d) Homoskedastic (diagonal $\sigma^2$) + no autocorrelation (off-diagonal 0) $=\sigma^2I$, so $\sigma^2\XXi X'X\XXi=\sigma^2\XXi$. $\blacksquare$}

\card{L9 Q3: LLN and CLT}
{$X'\veps=n\bar w$; LLN $\Rightarrow$ consistent; CLT $\Rightarrow$ normal.}
{(a) $b=\beta+(X'X/n)^{-1}\bar w$.\quad (b) LLN: $\bar w\to0$, so $b\to\beta$ (consistent).\\
(c) CLT: $\bar w$ is approximately normal, so $b\overset{a}{\sim}N(\beta,\sigma^2\XXi)$ even without normal errors.}

%===================== L10 =====================
\card{L10 Q1: $P_XH=0$ and $a=0$}
{$Xb-Xb$; then $b-b$.}
{(a) $P_X=X\XXi X'$.\quad (b) $P_X(Y-Xb)=Xb-Xb=0$.\\
(c) Residuals are orthogonal to every column of $X$.\quad (d) $a=\XXi X'(Y-Xb)=b-b=0$. $\blacksquare$}

\card{L10 Q2: projecting on a column of ones}
{every entry $1/n$, then $\bar x$, then $x_i-\bar x$.}
{(a) $P_Z$: $n\times n$, every entry $1/n$.\quad (b) $P_ZX$: $n\times1$, every entry $\bar x$.\quad (c) $X^*=M_ZX$: $n\times1$, entry $i$ is $x_i-\bar x$.}

%===================== L11 =====================
\card{L11 Q1: standardized estimate}
{$N(0,1)$ with $\sigma$; $t_{n-K}$ with $s$.}
{$\dfrac{b_k-\beta_k}{\sqrt{\sigma^2[\XXi]_{kk}}}\sim N(0,1)$; replacing $\sigma^2$ with $s^2=\dfrac{e'e}{n-K}$ gives $\dfrac{b_k-\beta_k}{\widehat{SE}(b_k)}\sim t_{n-K}$.}

\card{L11 Q2: $s^2$ is unbiased}
{$M'=M$, $MM=M$, $\tr M=n-K$.}
{(a) $e'e=\veps'M'M\veps=\veps'M\veps$ (symmetric: $M'=M$; idempotent: $MM=I-2P+P=M$).\\
(b) $\E(e'e|X)=\tr(M\sigma^2I)=\sigma^2[n-\tr(\XXi X'X)]=\sigma^2(n-K)$, so divide by $n-K$. $\blacksquare$}

\card{L11 Q3: bivariate $\Var(b_2)$}
{$(2,2)$ entry of $\XXi$; the determinant is $n\,SST_x$.}
{$\Var(b_2|X)=\dfrac{\sigma^2 n}{n\sum x_i^2-(\sum x_i)^2}=\dfrac{\sigma^2}{\sum(x_i-\bar x)^2}$.\quad Factors: $\sigma^2\uparrow$ worse; $n\uparrow$ better; spread in $x\uparrow$ better.}

\card{L11 Q4: multiple $\Var(b_3)$}
{$\sigma^2$ over $(1-R_3^2)\,SST_3$.}
{$\Var(b_3|X)=\dfrac{\sigma^2}{(1-R_3^2)SST_3}$. More precise with a smaller $\sigma^2$ (noise), a bigger $SST_3$ (variation, $n$), or a smaller $R_3^2$ (less correlation with the other $x$'s).}

\card{L11 Q5: multicollinearity}
{$R_k^2\uparrow\Rightarrow\Var\uparrow$; dropping the variable $\Rightarrow$ OVB.}
{(a) High $R_k^2$ makes $1-R_k^2\to0$, so the SE grows (still unbiased).\\
(b) Dropping $x_j$ lowers the variance, but $x_j$ goes into $\varepsilon$, so $\E(\varepsilon|X)\neq0$: omitted variable bias. You trade bias for variance.}

%===================== L12 =====================
\card{L12 Q1: log-level, $\log fare=4.666+0.043\,dist$}
{exponentiate; $\beta\cdot fare$; 1 mile $=0.01$.}
{$fare=e^{4.666+0.043dist}$;\; $\partial fare/\partial dist=0.043\,fare$;\; $100\tfrac{\partial fare}{fare}=4.3(0.01)=0.043\%$. $\blacksquare$}

\card{L12 Q2: level-log, $fare=41+66\log dist$}
{$\beta/x$; $\beta/100$ per 1\%; units only change the intercept.}
{$\partial fare/\partial dist=66/dist$;\; $\partial fare=\tfrac{66}{100}(\%\Delta dist)=\$0.66$ per 1\%.\\
Miles: $41+66\log(m/100)=(41-66\log100)+66\log m$, so the slope is unchanged. $\blacksquare$}

\card{L12 Q3: log-log, $\log fare=4.25+0.40\log dist$}
{elasticity: $\%\Delta y=\beta\,\%\Delta x$.}
{$fare=e^{4.25}dist^{0.40}$;\; $\partial fare/\partial dist=0.40\,fare/dist$;\; $\%\Delta fare=0.40\times10=4\%$. $\blacksquare$}
